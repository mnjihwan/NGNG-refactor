/** RF040 baseline: actual SecurityConfig/TokenFilter/JWT with a repository double, no Boot/network. */
class TokenStateSecurityChainTest {
    static AnnotationConfigWebApplicationContext context;
    static MockMvc mvc;
    static final AtomicInteger calls = new AtomicInteger();
    UserRepository users;
    User current;
    @BeforeAll static void start() {
        context = new AnnotationConfigWebApplicationContext();
        context.getEnvironment().getPropertySources().addFirst(new org.springframework.core.env.MapPropertySource(
                "rf040-only", Map.of("jwt.expiration.access", "3600000", "jwt.expiration.refresh", "7200000")));
        context.setServletContext(new MockServletContext()); context.register(Config.class); context.refresh();
        mvc = MockMvcBuilders.webAppContextSetup(context).apply(springSecurity()).build();
    }
    @AfterAll static void stop() { context.close(); }
    @BeforeEach void setup() {
        calls.set(0); SecurityContextHolder.clearContext(); users = context.getBean(UserRepository.class); reset(users);
        current = User.builder().userId(40L).email("rf005@example.test").visible(true).role(Role.builder().roleType("USER").build()).build();
        when(users.findUserByEmail("rf005@example.test")).thenAnswer(inv -> Optional.ofNullable(current));
    }
    @AfterEach void cleared() { assertNull(SecurityContextHolder.getContext().getAuthentication()); }
    @ParameterizedTest @ValueSource(strings = {"false", "null", "missing"})
    void beforeAccessIgnoresAccountState(String state) throws Exception {
        if (state.equals("missing")) current = null;
        else current.setVisible(state.equals("null") ? null : false);
        mvc.perform(get("/my_page/state").header("Authorization", "Bearer " + TokenFilterTest.token("access", "USER", 3600)))
                .andExpect(status().isOk()).andExpect(content().string("ROLE_USER"));
        assertEquals(1, calls.get()); verifyNoInteractions(users);
    }
    @Test void beforeAdminClaimKeepsAuthorityAfterDatabaseDowngrade() throws Exception {
        mvc.perform(get("/admin/state").header("Authorization", "Bearer " + TokenFilterTest.token("access", "ADMIN", 3600)))
                .andExpect(status().isOk()).andExpect(content().string("ROLE_ADMIN"));
        verifyNoInteractions(users);
    }
    @Test void beforeUserClaimDoesNotSeeDatabaseUpgrade() throws Exception {
        current.getRole().setRoleType("ADMIN");
        mvc.perform(get("/admin/state").header("Authorization", "Bearer " + TokenFilterTest.token("access", "USER", 3600)))
                .andExpect(status().isForbidden()); assertEquals(0, calls.get()); verifyNoInteractions(users);
    }
    @Test void beforeRefreshReissuesOldRoleForInactiveAccount() throws Exception {
        current.setVisible(false);
        var result = mvc.perform(get("/admin/state").cookie(new Cookie("refreshToken", TokenFilterTest.token("refresh", "ADMIN", 3600))))
                .andExpect(status().isOk()).andReturn();
        assertEquals("ADMIN", context.getBean(JwtTokenVerifier.class).getRole(result.getResponse().getHeader("Authorization")));
        verifyNoInteractions(users);
    }
    @Configuration @EnableWebMvc @EnableWebSecurity @Import(SecurityConfig.class)
    static class Config {
        @Bean @Primary AuthenticationConfiguration authenticationConfiguration() throws Exception {
            var c = mock(AuthenticationConfiguration.class); when(c.getAuthenticationManager()).thenReturn(mock(AuthenticationManager.class)); return c;
        }
        @Bean JwtTokenVerifier verifier() { return new JwtTokenVerifier(TokenFilterTest.SECRET); }
        @Bean JwtTokenProvider provider() { var p = spy(new JwtTokenProvider(TokenFilterTest.SECRET)); ReflectionTestUtils.setField(p, "accessTokenExpirationMs", 3_600_000); return p; }
        @Bean CustomLogoutHandler logoutHandler() { return new CustomLogoutHandler(); }
        @Bean CustomLogoutSuccessHandler logoutSuccessHandler() { return new CustomLogoutSuccessHandler(); }
        @Bean UserRepository users() { return mock(UserRepository.class); }
        @Bean Probe probe() { return new Probe(); }
    }
    @RestController static class Probe {
        @RequestMapping("/**") String state(Authentication auth) { calls.incrementAndGet(); return auth == null ? "anonymous" : auth.getAuthorities().iterator().next().getAuthority(); }
    }
}
