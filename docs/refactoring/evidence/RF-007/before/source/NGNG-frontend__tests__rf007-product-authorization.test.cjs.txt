// Runs actual API, hook and component handler code with fetch/React hook doubles and installed SWR cache.
// No browser, HTTP server, DB or new dependency. RF007_BEFORE=true records original behavior.
const test = require("node:test");
const assert = require("node:assert/strict");
const fs = require("node:fs");
const path = require("node:path");
const vm = require("node:vm");
const ts = require("typescript");
const react = require("react");
const { initCache } = require("swr/_internal");
const before = process.env.RF007_BEFORE === "true";
function load(relative, deps, globals = {}) {
  const filename = path.resolve(__dirname, "..", relative);
  const code = ts.transpileModule(fs.readFileSync(filename, "utf8"), {
    fileName: filename,
    compilerOptions: { target: ts.ScriptTarget.ES2020, module: ts.ModuleKind.CommonJS, jsx: ts.JsxEmit.ReactJSX, esModuleInterop: true },
  }).outputText;
  const module = { exports: {} };
  vm.runInNewContext(code, { module, exports: module.exports, ...globals, require(name) {
    if (name === "react/jsx-runtime") return require(name);
    if (Object.hasOwn(deps, name)) return deps[name];
    throw new Error(`Unexpected module: ${name}`);
  } }, { filename });
  return module.exports;
}
function response(status) { return new Response(status === 204 ? null : "77", { status }); }
function fixture(result) {
  const requests = [], tokens = [], alerts = [], moves = [], mutations = [];
  const auth = { getAccessToken: () => "original-token", setAccessToken: res => tokens.push(res) };
  const fetch = async (url, options) => { requests.push({ url, ...options }); if (result instanceof Error) throw result; return result; };
  const globals = { fetch, alert: message => alerts.push(message) };
  const updateApi = load("src/app/my_page/modify/_api/api.ts", {
    "@/app/admin/products/_api/api": {}, "../../_utils/auth-header": auth,
  }, globals);
  const deleteApi = load("src/app/my_page/_api/api.ts", { "../_utils/auth-header": auth }, globals);
  const initial = { sellList: [{ id: 77, visible: true }, { id: 88, visible: true }], buyList: [] };
  const [provider, mutate] = initCache(new Map()); const key = "/api/users";
  provider.set(key, { data: initial });
  const hook = load("src/app/my_page/_hooks/useMypageSWR.ts", {
    "../_api/api": deleteApi,
    swr: () => ({ data: initial, mutate: async (updater, options) => {
      mutations.push(options); return mutate(key, updater, options);
    } }),
  }, globals).default();
  const data = { id: 77, user: { id: 1000 }, title: "original", content: "old", visible: true,
    status: { id: 1 }, category: { id: 1 }, images: [], tags: [{ tagName: "old" }] };
  let stateIndex = 0, refIndex = 0;
  const states = [data, [], [], 1, 0, "new"];
  const refs = [{ value: "changed" }, { value: "3" }, { value: "2026-11" }, { value: "400" }, { checked: true }, { checked: true }, { value: "new content" }];
  const noop = () => null;
  const deps = Object.fromEntries([
    "@/components/common/inputs/CheckBox", "@/components/common/inputs/Input", "@/components/common/inputs/Radio",
    "next/image", "@/assets/SVG/Close.svg", "@/app/sell/_components/ImageUploadBox", "@/components/common/drop_down/DropDown",
  ].map(name => [name, noop]));
  Object.assign(deps, {
    react: { ...react, useState: () => [states[stateIndex++], noop], useEffect: noop, useRef: () => ({ current: refs[refIndex++] }) },
    "@/app/my_page/_utils/convert": { convertEnumToKeyValuesObject: () => [] },
    "@/app/my_page/_utils/extract": { imageUrlExtractExtension: noop },
    "../../_api/api": updateApi,
    "next/navigation": { useRouter: () => ({ replace: target => moves.push(target) }) },
  });
  const tree = load("src/app/my_page/modify/[productId]/_compoents/UpdateProductId.tsx", deps, globals).default({ productId: 77 });
  function findButton(node) {
    if (!node || typeof node !== "object") return undefined;
    if (node.type === "button" && node.props.children === "변경 하기") return node;
    return react.Children.toArray(node.props?.children).map(findButton).find(Boolean);
  }
  const button = findButton(tree); assert.ok(button);
  return { hook, update: button.props.onClick, deleteApi, updateApi, requests, tokens, alerts, moves, mutations, initial, cache: () => provider.get(key).data };
}
for (const status of [200, 401, 403, 500]) {
  test(`PUT ${status} component ${status === 200 || before ? "navigates" : "alerts and stays"}`, async () => {
    const f = fixture(response(status)); await f.update();
    assert.equal(f.moves.length, status === 200 || before ? 1 : 0);
    assert.equal(f.alerts.length, status === 200 || before ? 0 : 1);
    assert.equal(f.requests[0].url, "/api/products/77"); assert.equal(f.requests[0].method, "PUT");
    assert.equal(f.requests[0].headers.Authorization, "original-token");
    const body = JSON.parse(f.requests[0].body); assert.equal(body.title, "changed"); assert.equal(body.userId, 1000);
  });
}
test("PUT network failure is consumed by component after implementation", async () => {
  const f = fixture(new Error("offline"));
  if (before) await assert.rejects(f.update()); else await f.update();
  assert.equal(f.moves.length, 0); assert.equal(f.alerts.length, before ? 0 : 1);
});
for (const status of [200, 204, 401, 403, 500]) {
  test(`DELETE ${status} ${status === 200 || before ? "removes" : "preserves"} sale list`, async () => {
    const f = fixture(response(status)); await f.hook.deleteProduct(77); await new Promise(setImmediate);
    const accepted = status === 200 || before;
    assert.equal(f.mutations.length, accepted ? 1 : 0); assert.equal(f.alerts.length, accepted ? 0 : 1);
    assert.deepEqual(f.cache().sellList.map(p => p.id), accepted ? [88] : [77, 88]);
    if (!accepted) assert.equal(f.cache(), f.initial);
    assert.equal(f.requests[0].url, "/products/77"); assert.equal(f.requests[0].method, "DELETE");
    assert.equal(f.tokens.length, 1);
  });
}
test("DELETE network failure is consumed inside the hook", async () => {
  const f = fixture(new Error("offline"));
  if (before) await assert.rejects(f.hook.deleteProduct(77)); else await f.hook.deleteProduct(77);
  assert.equal(f.cache(), f.initial); assert.equal(f.mutations.length, 0); assert.equal(f.alerts.length, before ? 0 : 1);
});
if (!before) {
  test("unawaited DELETE failure cannot produce an unhandled rejection", async () => {
    const f = fixture(response(403)); const unhandled = [];
    const listener = reason => unhandled.push(reason); process.on("unhandledRejection", listener);
    try {
      // Matches my_page's callback: the caller discards the promise.
      f.hook.deleteProduct(77); await new Promise(setImmediate); await new Promise(setImmediate);
      assert.equal(unhandled.length, 0); assert.equal(f.alerts.length, 1); assert.equal(f.mutations.length, 0); assert.equal(f.cache(), f.initial);
    } finally { process.removeListener("unhandledRejection", listener); }
  });
}
